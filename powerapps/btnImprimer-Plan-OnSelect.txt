// btnImprimer.OnSelect (onglet Plan) — PDF du plan de développement
// 1) assemble la page HTML dans varHtmlPDF ; 2) le flux GenererPDFAppreciation la convertit
// en PDF et la joint au dossier (liste Appreciations) en remplaçant la version précédente ;
// 3) le bouton « 📄 Ouvrir le PDF » (ou Mes dossiers) l'ouvre.
If(IsBlank(varDossier);
    Notify("Enregistrez d'abord le dossier avec « Enregistrer et déposer »."; NotificationType.Warning);
    Set(varLienPDF; Blank());;
    Notify("Création du PDF en cours… Comptez de 10 à 30 secondes."; NotificationType.Information);;
    Set(varHtmlPDF; "<html><head><meta charset='utf-8'><style>body{font-family:Segoe UI,Arial,sans-serif;font-size:10pt;color:#1f2937}h1{font-size:18pt;color:#00285e;margin:0}.sous{color:#6b7280;margin:2px 0 14px 0}h2{font-size:12.5pt;color:#00285e;border-bottom:2px solid #857650;padding-bottom:3px;margin:18px 0 6px 0}h3{font-size:11pt;color:#074b98;margin:12px 0 4px 0}table{border-collapse:collapse;width:100%;margin:4px 0 8px 0}th{background:#e8eef7;color:#00285e;text-align:left;padding:5px;border:1px solid #c7d2e3;font-size:9.5pt}td{padding:5px;border:1px solid #d1d5db;vertical-align:top;font-size:9.5pt}td.lbl{background:#f3f4f6;font-weight:bold;width:28%}.vide{color:#9ca3af;font-style:italic}.desc{color:#4b5563;font-style:italic;margin:0 0 4px 0}.bloc{border:1px solid #d1d5db;padding:8px;min-height:30px}.pied{margin-top:24px;color:#9ca3af;font-size:8pt;border-top:1px solid #e5e7eb;padding-top:4px}</style></head><body>");;
    Set(varHtmlPDF; varHtmlPDF & "<h1>Plan de développement</h1><p class=sous>" & LookUp(colPostes; PosteID = CurrentPosteID).PosteLabel & " — dossier no " & varDossier.ID & "</p><table>");;
    Set(varHtmlPDF; varHtmlPDF & "<tr><td class=lbl>Nom, prénom</td><td>" & Substitute(Substitute(Substitute(Substitute(Coalesce(txtNom.Text; varDossier.Title); "&"; "&amp;"); "<"; "&lt;"); ">"; "&gt;"); Char(10); "<br>") & "</td></tr>");;
    Set(varHtmlPDF; varHtmlPDF & "<tr><td class=lbl>Poste</td><td>" & Substitute(Substitute(Substitute(Substitute(Coalesce(txtPoste.Text; varDossier.Poste); "&"; "&amp;"); "<"; "&lt;"); ">"; "&gt;"); Char(10); "<br>") & "</td></tr>");;
    Set(varHtmlPDF; varHtmlPDF & "<tr><td class=lbl>Supérieur(e) immédiat(e)</td><td>" & Coalesce(varDossier.SuperieurImmediat.DisplayName; "—") & "</td></tr>");;
    Set(varHtmlPDF; varHtmlPDF & "<tr><td class=lbl>Année</td><td>" & Coalesce(Text(varDossier.Annee); Text(Year(Today()))) & "</td></tr>");;
    Set(varHtmlPDF; varHtmlPDF & "</table><h2>Objectifs pour l'année à venir</h2>");;
    Set(varHtmlPDF; varHtmlPDF & If(CountRows(Filter(colObjAvenir; Objectif <> "" Or Indicateur <> "")) = 0; "<p class=vide>Aucun élément.</p>"; "<table><tr><th>Objectifs à venir</th><th>Indicateurs de mesure</th></tr>" & Concat(Filter(colObjAvenir; Objectif <> "" Or Indicateur <> ""); "<tr><td>" & Substitute(Substitute(Substitute(Substitute(Objectif; "&"; "&amp;"); "<"; "&lt;"); ">"; "&gt;"); Char(10); "<br>") & "</td><td>" & Substitute(Substitute(Substitute(Substitute(Indicateur; "&"; "&amp;"); "<"; "&lt;"); ">"; "&gt;"); Char(10); "<br>") & "</td></tr>") & "</table>"));;
    Set(varHtmlPDF; varHtmlPDF & "<h2>Développement des compétences — résultats des actions réalisées</h2>");;
    Set(varHtmlPDF; varHtmlPDF & If(CountRows(Filter(colDevPrecedents; Objectif <> "" Or Moyens <> "" Or Impacts <> "")) = 0; "<p class=vide>Aucun élément.</p>"; "<table><tr><th>Objectifs précédents</th><th>Moyens choisis</th><th>Impacts / résultats</th></tr>" & Concat(Filter(colDevPrecedents; Objectif <> "" Or Moyens <> "" Or Impacts <> ""); "<tr><td>" & Substitute(Substitute(Substitute(Substitute(Objectif; "&"; "&amp;"); "<"; "&lt;"); ">"; "&gt;"); Char(10); "<br>") & "</td><td>" & Substitute(Substitute(Substitute(Substitute(Moyens; "&"; "&amp;"); "<"; "&lt;"); ">"; "&gt;"); Char(10); "<br>") & "</td><td>" & Substitute(Substitute(Substitute(Substitute(Impacts; "&"; "&amp;"); "<"; "&lt;"); ">"; "&gt;"); Char(10); "<br>") & "</td></tr>") & "</table>"));;
    Set(varHtmlPDF; varHtmlPDF & "<h2>Nouveaux besoins et moyens de développement</h2>");;
    Set(varHtmlPDF; varHtmlPDF & If(CountRows(Filter(colDevNouveaux; Besoin <> "" Or Moyen <> "")) = 0; "<p class=vide>Aucun élément.</p>"; "<table><tr><th>Besoins de développement</th><th>Moyens de développement</th></tr>" & Concat(Filter(colDevNouveaux; Besoin <> "" Or Moyen <> ""); "<tr><td>" & Substitute(Substitute(Substitute(Substitute(Besoin; "&"; "&amp;"); "<"; "&lt;"); ">"; "&gt;"); Char(10); "<br>") & "</td><td>" & Substitute(Substitute(Substitute(Substitute(Moyen; "&"; "&amp;"); "<"; "&lt;"); ">"; "&gt;"); Char(10); "<br>") & "</td></tr>") & "</table>"));;
    Set(varHtmlPDF; varHtmlPDF & "<p class=pied>Document produit le " & Text(Now(); "d mmmm yyyy") & " à " & Text(Now(); "hh:mm") & " à partir de l'application Appréciation de la contribution.</p></body></html>");;
    Set(varLienPDF;
        GenererPDFAppreciation.Run(
            "Plan_developpement_dossier" & varDossier.ID;
            varHtmlPDF;
            varDossier.ID).lien);;
    If(IsBlank(varLienPDF);
        Notify("Le PDF n'a pas pu être créé. Réessayez dans un moment."; NotificationType.Error);
        Notify("PDF prêt : cliquez sur « 📄 Ouvrir le PDF »."; NotificationType.Success)))
