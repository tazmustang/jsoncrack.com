// Recopie les tableaux de la section A (écran -> collections), sans faire sauter le curseur

    {LigneID: LigneID; Objectif: txtActObjectif.Text; Indicateur: txtActIndicateur.Text; Actions: txtActActions.Text}));;

    {LigneID: LigneID; Objectif: txtAveObjectif.Text; Indicateur: txtAveIndicateur.Text}));;

        Notify("Ouvrez d'abord le dossier de la personne depuis « Mes dossiers »."; NotificationType.Warning);
        With({si: IfError('Utilisateursd’Office365'.ManagerV2(User().Email); Blank())};
        With({sh: IfError('Utilisateursd’Office365'.ManagerV2(si.mail); Blank())};
            Set(varDossier; Patch(Appreciations; Defaults(Appreciations); {
                Titre: txtNom.Text;
                Annee: Year(Today());
                VersionFormulaire: "15 octobre 2026";
                Statut: {Value: "Brouillon"};
                PersonneAppreciation: {Claims: "i:0#.f|membership|" & varMoi;
                    DisplayName: User().FullName; Email: User().Email;
                    Department: ""; JobTitle: ""; Picture: ""};
                CourrielPersonne: varMoi;
                SuperieurImmediat: If(!IsBlank(si.mail); {Claims: "i:0#.f|membership|" & Lower(si.mail);
                    DisplayName: si.displayName; Email: si.mail;
                    Department: ""; JobTitle: ""; Picture: ""});
                CourrielSI: Lower(si.mail);
                SuperieurHierarchique: If(!IsBlank(sh.mail); {Claims: "i:0#.f|membership|" & Lower(sh.mail);
                    DisplayName: sh.displayName; Email: sh.mail;
                    Department: ""; JobTitle: ""; Picture: ""});
                CourrielSH: Lower(sh.mail)
            }))))));;

If(!IsBlank(varDossier);
    Set(varDossier; Patch(Appreciations; varDossier; {
        Titre: txtNom.Text;
        IdEmploye: txtId.Text;
        Poste: Coalesce(txtPoste.Text; varDossier.Poste; 'Utilisateursd’Office365'.MyProfileV2().jobTitle);
        TypePoste: If(CurrentPosteID = ""; Blank();
            {Value: LookUp(colPostes; PosteID = CurrentPosteID).PosteLabel});
        TypeRencontre: Filter(Choices(Appreciations.TypeRencontre);
            ("essai" in Value And chkEssai.Checked) Or
            ("premi" in Value And chkPremiere.Checked) Or
            ("mi-ann" in Value And chkMiAnnee.Checked) Or
            ("bilan annuel" in Value And chkBilan.Checked) Or
            (StartsWith(Value; "Autre") And (chkAutre.Checked Or !IsBlank(Trim(txtAutre.Text)))));
        TypeRencontreAutre: txtAutre.Text;
        ObjectifsActuels: JSON(ForAll(colObjActuels;
            {objectif: Objectif; indicateur: Indicateur; actions: Actions}));
        ObjectifsAvenir: JSON(ForAll(colObjAvenir;
            {objectif: Objectif; indicateur: Indicateur}));
        Postures: JSON({
            cotes: ForAll(colComportements; {k: CompKey; cadre: CoteCadre; sup: CoteSup});
            commentaires: ForAll(colPostures; {p: PostureID; texte: Commentaire})});
        Competences: JSON(ForAll(colCompetences; {id: CompID; exemple: Exemple}));
        Valeurs: JSON(ForAll(colValeurs; {s: Slug; coche: Coche; ex1: Ex1; ex2: Ex2}));
        DevPrecedents: JSON(ForAll(colDevPrecedents; {objectif: Objectif; moyens: Moyens; impacts: Impacts}));
        DevNouveaux: JSON(ForAll(colDevNouveaux; {besoin: Besoin; moyen: Moyen}));
        BilanSuperieur: txtBilanSup.Text;
        BilanPersonne: txtBilanPersonne.Text;
        EvaluationGlobale: If(varEvalGlobale = ""; Blank(); {Value: varEvalGlobale});
        DirecteurRequis: chkDirRequis.Checked;
        Directeur: If(!IsBlank(cboDirecteur.Selected);
            {Claims: "i:0#.f|membership|" & Lower(cboDirecteur.Selected.Mail);
             DisplayName: cboDirecteur.Selected.DisplayName; Email: cboDirecteur.Selected.Mail;
             Department: ""; JobTitle: ""; Picture: ""};
            varDossier.Directeur);
        CourrielDirecteur: If(!IsBlank(cboDirecteur.Selected);
            Lower(cboDirecteur.Selected.Mail); varDossier.CourrielDirecteur)
    }));;
    If(IsEmpty(Errors(Appreciations));
        Notify("Dossier enregistré."; NotificationType.Success);
        Notify("Erreur : " & First(Errors(Appreciations)).Message; NotificationType.Error)))

// btnDeposer.OnSelect — « Enregistrer et déposer »
// 1) Personne en appréciation : vérifie les champs obligatoires ; 2) nouveau dossier : demande une confirmation ;
// 3) sinon : enregistre (btnDeposerExec)
If(
    // Champs obligatoires vides (seulement pour la personne en appréciation, qui les remplit)
    CurrentRoleID = "cadre" && lblManquants.Text <> "";
    Notify("Avant d'enregistrer, complétez : " & lblManquants.Text; NotificationType.Warning);
    // Nouveau dossier : on demande la confirmation de création
    IsBlank(varDossier);
    UpdateContext({locConfirmer: true});
    // Dossier existant : on enregistre directement
    Select(btnDeposerExec)
)
