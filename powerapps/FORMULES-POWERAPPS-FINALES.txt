NOTE : « fr-CA » n'est PAS accepté dans votre environnement, ni comme 3e argument de Text(…),
ni dans les formats de date « [$-fr-CA]… ». Les deux provoquent l'erreur « Code de langue fr-CA
non pris en charge ». Écrivez simplement Text(date; "d mmmm yyyy") : la date s'affiche dans la
langue de la personne (en français pour les employés de la Ville). Voir la partie 9.

galEtapes.Items
With(
    {S: Coalesce(varDossier.Statut.Value; "Brouillon");
     dirReq: Coalesce(varDossier.DirecteurRequis; false);
     d2: !IsBlank(varDossier.CertifieLe);
     d3: !IsBlank(varDossier.ApprouveSILe);
     d4: !IsBlank(varDossier.ApprouveSHLe);
     d5: !IsBlank(varDossier.ApprouveDirLe);
     d6: !IsBlank(varDossier.TransmisPaieLe)};
    Table(
        {Ordre: 1; Etape: "Brouillon"; Fait: true; EnCours: S = "Brouillon" Or S = "Avisé au supérieur"; Saute: false; DateEtape: If(false; Now()); SuivantFait: d2};
        {Ordre: 2; Etape: "Certifié (personne en appréciation)"; Fait: d2; EnCours: false; Saute: false; DateEtape: varDossier.CertifieLe; SuivantFait: d3};
        {Ordre: 3; Etape: "Approuvé (sup. immédiat)"; Fait: d3; EnCours: S = "Certifié"; Saute: false; DateEtape: varDossier.ApprouveSILe; SuivantFait: d4};
        {Ordre: 4; Etape: "Approuvé (sup. hiérarchique)"; Fait: d4; EnCours: S = "Approuvé SI"; Saute: false; DateEtape: varDossier.ApprouveSHLe; SuivantFait: d5};
        {Ordre: 5; Etape: "Approuvé (directeur)"; Fait: d5; EnCours: S = "Approuvé SH"; Saute: Not(dirReq); DateEtape: varDossier.ApprouveDirLe; SuivantFait: d6};
        {Ordre: 6; Etape: "Transmis aux RH"; Fait: d6; EnCours: S = "Complété" And Not(d6); Saute: false; DateEtape: varDossier.TransmisPaieLe; SuivantFait: S = "Complété"};
        {Ordre: 7; Etape: "Archivé (lecture seule)"; Fait: S = "Complété"; EnCours: false; Saute: false; DateEtape: If(false; Now()); SuivantFait: false}))
